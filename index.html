from src.predict import predict_signal_level

# Execute inference for a mobile user equipment connection
result = predict_signal_level(
    frequency_mhz=2100.0,
    distance_m=650.0,
    tx_power_dbm=40.0,
    tx_gain_dbi=16.0,
    rx_gain_dbi=2.0,
    environment="Urban",
    obstacle_count=4,
    humidity_pct=65.0,
    weather_condition="Rain"
)

print(f"Predicted Signal: {result['predicted_signal_level']}")  # e.g. 'Poor' / 'Fair' / 'Good'
print(f"Confidence:       {result['confidence']:.2%}")
print(f"Probabilities:    {result['probabilities']}")
